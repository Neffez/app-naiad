  #include <abstractions/base>

  # Capabilities
  capability,

  # Network access (HA Supervisor proxy + WebSocket to Core)
  network,

  # Allow the bootstrap shell and the Python interpreter to run
  /run.sh r,
  /bin/** rix,

  /usr/local/bin/** rix,

  # Application code and the Python runtime (read-only)
  /app/** r,

  /usr/lib/** mr,
  /lib/** mr,

  # Persisted data volume: SQLite database + optional config.yaml seed + options.json
  /data/ rw,

  # Scratch space
  /tmp/ rw,
  /tmp/** rwk,

  # proc/sys bits commonly read by the runtime
  /proc/sys/kernel/random/uuid r,
  /sys/kernel/mm/transparent_hugepage/enabled r,
  @{PROC}/@{pid}/** r,
}
